lvm合并两块物理磁盘

合并/dev/sda和/dev/sdb

安装lvm2工具包(如果尚未安装)

sudo apt install lvm2   # Debian/Ubuntu系统
# 或
sudo yum install lvm2   # RHEL/CentOS系统

将硬盘格式化为LVM格式：

# 使用fdisk创建LVM分区
sudo fdisk /dev/sda
# 输入n新建分区
# 输入p创建主分区
# 分区号使用默认的1
# 其他选项使用默认值
# 输入t更改分区类型
# 输入8e选择Linux LVM类型
# 输入w保存更改

# 对sdb重复相同操作
sudo fdisk /dev/sdb

创建物理卷(PV):

sudo pvcreate /dev/sda1
sudo pvcreate /dev/sdb1

# 检查物理卷
sudo pvs

创建卷组(VG):

# 创建名为vg_data的卷组，并将第一个物理卷添加进去
sudo vgcreate vg_data /dev/sda1

# 将第二个物理卷扩展到卷组中
sudo vgextend vg_data /dev/sdb1

# 检查卷组
sudo vgs
创建逻辑卷(LV):

# 创建使用全部空间的逻辑卷
sudo lvcreate -l 100%VG -n lv_data vg_data

# 检查逻辑卷
sudo lvs
格式化逻辑卷：

# 使用ext4文件系统格式化
sudo mkfs.ext4 /dev/vg_data/lv_data

创建挂载点并挂载：

# 创建挂载点
sudo mkdir /data

# 挂载逻辑卷
sudo mount /dev/vg_data/lv_data /data

# 检查挂载情况
df -h
设置开机自动挂载：

# 编辑/etc/fstab文件
sudo nano /etc/fstab

# 添加以下行
/dev/vg_data/lv_data /data ext4 defaults 0 0

B盘合并至有数据的A盘(parted)

/dev/sdc 合并至 /dev/sdb

模拟数据，对/dev/sdb格式化并写入测试数据

# 分区
parted /dev/sdb 
(parted) mklabel gpt            # GPT（即GUID分区表） ， 突破MBR 4个主分区限制
(parted) mkpart p 0% 100%       # 主分区
(parted) print
(parted) toggle 1 lvm           # 将分区打上lvm标签 
(parted) print               
(parted) q
partprobe    # 重读分区表

# 创建pv、vg和lv
pvcreate /dev/sdb1
pvs
vgcreate vg_data /dev/sdb1
vgs
sudo lvcreate -l 100%VG -n lv_data vg_data
lvs
mkfs.xfs /dev/vg_data/lv_data

# 挂载
mkdir /data
mount /dev/vg_data/lv_data /data/

格式化 /dev/sdc

# 分区
parted /dev/sdc
(parted) mklabel gpt            # GPT（即GUID分区表） ， 突破MBR 4个主分区限制
(parted) mkpart p 0% 100%       # 主分区
(parted) print
(parted) toggle 1 lvm           # 将分区打上lvm标签 
(parted) print               
(parted) q
partprobe    # 重读分区表

pvcreate /dev/sdc1                # 创建pv
vgextend vg_data /dev/sdc1        # 把sdc添加上此前创建的vg中
lvextend -l +100%FREE /dev/mapper/vg_data-lv_data    # 扩展lv
resize2fs /dev/mapper/vg_data-lv_data   # 在线对逻辑卷进行调整

还原lvm合并的磁盘

  1. 卸载逻辑卷
# 确保没有进程在使用该挂载点
sudo fuser -km /data
# 卸载逻辑卷
sudo umount /data
  1. /etc/fstab中删除自动挂载项
# 删除包含目标逻辑卷的数据行
  1. 删除逻辑卷(LV)
# 查看逻辑卷
sudo lvs
# 删除逻辑卷
sudo lvremove /dev/vg_data/lv_data
# 确认删除时输入y
  1. 删除卷组(VG)
# 查看卷组
sudo vgs
# 删除卷组
sudo vgremove vg_data
  1. 删除物理卷(PV)
# 查看物理卷
sudo pvs
# 删除物理卷
sudo pvremove /dev/sda1
sudo pvremove /dev/sdb1
  1. 清除硬盘分区表
# 对sda进行操作
sudo fdisk /dev/sda
# 输入d删除分区
# 输入w保存并退出

# 对sdb进行操作
sudo fdisk /dev/sdb
# 输入d删除分区
# 输入w保存并退出

GPT PMBR size mismatch will be corrected by write错误

原因：在对虚拟机扩容时候，由于linux系统没有对其磁盘信息进行更新，导致了磁盘实际容量和linux系统容量不一致 修复

# 执行命令：
sudo parted -l

Fix

lvextend之后发现扩容的文件系统不对，需要还原

根分区有1.75TB实际可用空间，需要把多余的两块8T的盘加入根分区，在lvextend之后发现原来8T的盘被格式化为ext4，然后根分区是xfs文件系统，这会导致xfs_growfs resizefs的时候报错，所以需要把加入的两个pv，分别是/dev/sda1和/dev/sdb1给删掉。

修复
