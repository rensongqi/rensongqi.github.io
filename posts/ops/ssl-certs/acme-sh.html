安装acme.sh工具

git clone https://gitee.com/neilpang/acme.sh.git
cd acme.sh
./acme.sh --install -m songqi.ren@rensongqi.com

安装socat

# Centos
yum -y install socat

# Ubuntu
apt -y install socat

配置CA

./acme.sh --set-default-ca --server letsencrypt
